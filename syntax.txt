// --- Directives ---
@output json;                      // Specify output format
@output csv { delimiter = ";"; } // Output config block
@seed 42;                          // Seed random generator
@locale "en_US";                  // Set locale

// --- External Resource Loading ---
@load firstNames from "data/first_names.txt";
@load cities from "data/cities.csv" column 0;

// --- Resource Definition Block ---
resource Names {
    first = ["Ana", "Marko", "Ivana"];
    last  = ["Jovanović", "Petrović", "Nikolić"];
}

// --- Template Definition ---
template User(role: string) {
    id: string = uuid();
    name: string = pick(Names.first) + " " + pick(Names.last);
    role: string = role;
}

template Point {
    x: float(0..=100);
    y: float(0..=100);
    z: float(0..=100);
}

// --- Enum Declaration ---
enum Status {
    Active,
    Inactive,
    Pending
}

// --- User-Defined Function ---
fn full_phone() -> string {
    return format("+1-{0}-{1}", int(100,999), int(1000,9999));
}

// --- Data Generation ---
generate _ [10] {
    email: string = $format("{0}@company.com", slug(name));
    role: Role = $pick(Role);
    status: Status = Status.Active;
}

generate TemplateName [10];

// --- Assignment & Expressions ---
id: string = uuid();
name: string = randomName();
age: int = randomInt(18, 60);
isAdult: bool = age >= 18;

// --- Control Flow ---
if (age > 30) {
    status: string = "Senior";
} else {
    status: string = "Junior";
}

// --- Nesting Blocks ---
address {
    street: string = randomStreet();
    city: string = randomCity();
}

// --- List Generation ---
tags: list<string> = list(3) {
    randomTag();
};
